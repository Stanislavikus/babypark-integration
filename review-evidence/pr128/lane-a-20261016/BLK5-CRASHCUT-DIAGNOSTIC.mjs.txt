import test from 'node:test';
import assert from 'node:assert/strict';
import fs from 'node:fs';import os from 'node:os';import path from 'node:path';
import '../helpers/legacy-sending-fixture.mjs';
import { FirstLineStateStore } from '../../src/copilot/first-line-state-store.mjs';
import { prepareTestPublicAction,testActionDescriptor } from '../helpers/first-line-action-descriptor.mjs';
const NOW=2_000_000_000_000;
const msg=id=>({sourceMessageId:id,eventKind:'CUSTOMER_MESSAGE',messageType:'incoming',senderClass:'contact',senderId:9001,contentType:'text',deleted:false,unsupported:false,hasAttachments:false,sourceId:null});
const tables=['conversation_streams','conversation_events','semantic_origins','continuation_owners','public_actions','deferred_event_parents','public_action_human_cuts','episodes'];
function digest(store){return Object.fromEntries(tables.map(n=>[n,store.db.prepare('SELECT * FROM '+n+' ORDER BY rowid').all()]));}
function make(t,state){
 const dir=fs.mkdtempSync(path.join(os.tmpdir(),'bp-crash-matrix-'));t.after(()=>fs.rmSync(dir,{recursive:true,force:true}));
 const file=path.join(dir,'state.sqlite');
 let store=FirstLineStateStore.create(file,{now:()=>NOW});
 if(state==='EMPTY')return {file,store,action:null,stream:null};
 const stream=store.ensureConversationStream({sourceProvider:'chatwoot',sourceConversationId:55});
 store.ingestConversationEvent(stream.stream_id,msg(101));
 if(state==='NO_OWNER')return {file,store,action:null,stream};
 if(state==='HUMAN'){
  store.commitDirectHumanOrigin({streamId:stream.stream_id,streamRevision:1,reason:'initial_human'});
  return {file,store,action:null,stream};
 }
 const episode=store.beginEpisode({streamId:stream.stream_id});
 const action=prepareTestPublicAction(store,{descriptor:testActionDescriptor(),streamId:stream.stream_id,
  episodeId:episode.episode_id,expectedEpisodeVersion:episode.version,
  preparedStreamRevision:1,actionType:'CLARIFY',
  requestedSlot:'store_id',
  presentedCandidates:[{slot:'store_id',value:'store_11111111-1111-4111-8111-111111111111'}],
  basisEventSeqs:[1],deadlineAt:NOW+60000});
 if(['GATING','SENDING','UNCERTAIN'].includes(state))store.claimNextPublicAction({leaseMs:30000,token:'crash-lease'});
 if(['SENDING','UNCERTAIN'].includes(state))store.markActionSending(action.action_id,'crash-lease');
 if(state==='UNCERTAIN')store.markActionUncertain(action.action_id);
 return {file,store,action,stream};
}
const cases=[
 {state:'EMPTY',id:102,trigger:'BEFORE INSERT ON conversation_events'},
 {state:'EMPTY',id:102,trigger:'BEFORE INSERT ON semantic_origins'},
 {state:'NO_OWNER',id:102,trigger:'BEFORE INSERT ON semantic_origins'},
 {state:'PREPARED',id:102,trigger:'BEFORE INSERT ON semantic_origins'},
 {state:'PREPARED',id:102,trigger:'BEFORE UPDATE OF state ON public_actions'},
 {state:'GATING',id:102,trigger:'BEFORE UPDATE OF state ON public_actions'},
 {state:'PREPARED',id:101,trigger:'BEFORE UPDATE OF owner_kind ON continuation_owners'},
 {state:'GATING',id:101,trigger:'BEFORE UPDATE OF owner_kind ON continuation_owners'},
 {state:'SENDING',id:102,trigger:'BEFORE INSERT ON deferred_event_parents'},
 {state:'UNCERTAIN',id:102,trigger:'BEFORE INSERT ON deferred_event_parents'},
 {state:'SENDING',id:102,trigger:'BEFORE INSERT ON public_action_human_cuts'},
 {state:'UNCERTAIN',id:102,trigger:'BEFORE INSERT ON public_action_human_cuts'},
 {state:'HUMAN',id:102,trigger:'BEFORE INSERT ON conversation_events'},
];
for(const [i,config] of cases.entries()){
 test('BLK5 crash-cut SQL transaction '+i+' '+config.state+' '+config.trigger,t=>{
  const f=make(t,config.state);
  const before=digest(f.store);
  f.store.db.exec('CREATE TEMP TRIGGER fault_'+i+' '+config.trigger+" BEGIN SELECT RAISE(ABORT,'synthetic-cut-"+i+"'); END");
  assert.throws(()=>f.store.ingestVerifiedChatwootEvent({sourceConversationId:55,event:msg(config.id),
    humanReason:config.state==='HUMAN'?null:'NATIVE_CUSTOMER_UPDATE'}),new RegExp('synthetic-cut-'+i));
  assert.deepEqual(digest(f.store),before,'atomic rollback before close');
  f.store.close();
  f.store=FirstLineStateStore.open(f.file,{now:()=>NOW});
  assert.deepEqual(digest(f.store),before,'atomic rollback after restart');
  const retried=f.store.ingestVerifiedChatwootEvent({sourceConversationId:55,event:msg(config.id),
    humanReason:config.state==='HUMAN'?null:'NATIVE_CUSTOMER_UPDATE'});
  assert.equal(retried.inserted,config.id!==101);
  if(config.state!=='HUMAN'){
    const owner=f.store.getUnresolvedContinuationOwner(retried.stream.stream_id);
    assert.equal(owner?.owner_kind,'HUMAN');
  }
  f.store.close();
 });
}
